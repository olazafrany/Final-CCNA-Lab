==========================================
ISP1 Router Configuration
==========================================

enable
configure terminal
hostname ISP1

 --- Serial Interface to Branch ---
interface serial 0/1/0
 ip address 40.1.1.2 255.255.255.252
 clock rate 64000
 no shutdown
 exit

 --- Static Routes to NAT Public IPs ---
ip route 100.100.100.100 255.255.255.255 serial 0/1/0
ip route 8.8.8.8 255.255.255.255 serial 0/1/0

 --- Interface to LAN (HSRP) ---
interface gigabitethernet 0/0/0
 ip address 42.2.2.1 255.255.255.0
 standby 1 ip 42.2.2.3
 no shutdown
 exit

 --- Save Configuration ---
end
write memory
